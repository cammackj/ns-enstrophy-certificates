def read_json(path: Path) -> dict[str, Any]:
    return json.loads(path.read_text(encoding="utf-8"))


def read_csv(path: Path) -> list[dict[str, str]]:
    with path.open(newline="", encoding="utf-8") as handle:
        return list(csv.DictReader(handle))


def shell_support(row: dict[str, str]) -> tuple[int, int, int]:
    return tuple(sorted(int(row[key]) for key in ("output_shell", "first_input_shell", "second_input_shell")))


def status_for_group(face_class: str, residual_band: str, in_k34_signature_ledger: bool) -> str:
    if face_class == "k3_endpoint_face":
        return "covered_by_U3_endpoint_chart"
    if face_class == "k4_endpoint_safe_face":
        return "covered_inside_U4_chart_after_U4_radius_is_fixed"
    if face_class == "k3_k4_skeleton_bridge_face":
        return "open_skeleton_bridge_cover"
    if face_class == "window_residual_face" and not in_k34_signature_ledger:
        return "outside_k34_signature_rows_support_level_selector_exclusion_still_required"
    if face_class == "window_residual_face" and residual_band == "mixed_i3_i4_residual":
        return "open_mixed_residual_leakage_or_involution_classification"
    if face_class == "window_residual_face":
        return "open_single_band_residual_leakage_or_support_certificate"
    return "unclassified"


def build_face_ledger(rows: list[dict[str, str]]) -> list[dict[str, Any]]:
    groups: dict[tuple[str, str, str, bool], dict[str, Any]] = {}
    support_sets: dict[tuple[str, str, str, bool], set[tuple[int, int, int]]] = defaultdict(set)
    signature_sets: dict[tuple[str, str, str, bool], Counter[str]] = defaultdict(Counter)

    for row in rows:
        in_k34_signature_ledger = row["in_signature_ledger"] == "True"
        key = (
            row["face_class"],
            row["residual_band"],
            row["outside_skeleton_slot_count"],
            in_k34_signature_ledger,
        )
        record = groups.setdefault(
            key,
            {
                "face_class": row["face_class"],
                "residual_band": row["residual_band"],
                "outside_skeleton_slot_count": int(row["outside_skeleton_slot_count"]),
                "in_k34_signature_ledger": in_k34_signature_ledger,
                "ordered_triad_count": 0,
                "effective_ordered_triad_count": 0,
                "ordered_shell_face_rows": 0,
            },
        )
        record["ordered_triad_count"] += int(row["ordered_triad_count"])
        record["effective_ordered_triad_count"] += int(row["effective_ordered_triad_count"])
        record["ordered_shell_face_rows"] += 1
        support_sets[key].add(shell_support(row))
        signature_sets[key][row["ordered_signature"]] += int(row["effective_ordered_triad_count"])

    ledger = []
    for key, record in groups.items():
        face_class, residual_band, _outside, in_k34_signature_ledger = key
        top_supports = sorted(
            support_sets[key],
            key=lambda support: (
                -sum(
                    int(row["effective_ordered_triad_count"])
                    for row in rows
                    if shell_support(row) == support
                    and row["face_class"] == face_class
                    and row["residual_band"] == residual_band
                    and (row["in_signature_ledger"] == "True") == in_k34_signature_ledger
                ),
                support,
            ),
        )[:8]
        record["support_count"] = len(support_sets[key])
        record["top_supports"] = [list(support) for support in top_supports]
        record["top_signatures"] = [
            {"signature": sig, "effective_ordered_triad_count": count}
            for sig, count in signature_sets[key].most_common(8)
        ]
        record["cover_status"] = status_for_group(face_class, residual_band, in_k34_signature_ledger)
        ledger.append(record)

    return sorted(
        ledger,
        key=lambda row: (
            row["cover_status"].startswith("covered"),
            row["face_class"],
            row["residual_band"],
            row["outside_skeleton_slot_count"],
            row["in_k34_signature_ledger"],
        ),
    )


def build_mixed_support_ledger(rows: list[dict[str, str]]) -> list[dict[str, Any]]:
    selector_set = set(SELECTOR_SIGNATURES)
    supports: dict[tuple[int, int, int], dict[str, Any]] = {}
    signature_sets: dict[tuple[int, int, int], Counter[str]] = defaultdict(Counter)
    selected_signature_sets: dict[tuple[int, int, int], Counter[str]] = defaultdict(Counter)

    for row in rows:
        if row["face_class"] != "window_residual_face" or row["residual_band"] != "mixed_i3_i4_residual":
            continue
        effective_count = int(row["effective_ordered_triad_count"])
        if effective_count == 0:
            continue
        support = shell_support(row)
        record = supports.setdefault(
            support,
            {
                "support": " ".join(str(value) for value in support),
                "effective_ordered_triad_count": 0,
                "ordered_shell_face_rows": 0,
                "selected_family_effective_triad_count": 0,
                "has_selected_family_signature": False,
            },
        )
        record["effective_ordered_triad_count"] += effective_count
        record["ordered_shell_face_rows"] += 1
        signature_sets[support][row["ordered_signature"]] += effective_count
        if row["ordered_signature"] in selector_set:
            record["selected_family_effective_triad_count"] += effective_count
            record["has_selected_family_signature"] = True
            selected_signature_sets[support][row["ordered_signature"]] += effective_count

    endpoint_gap_if_in_strip = K4_ENDPOINT_THRESHOLD - A3_G4_UPPER
    ledger = []
    for support, record in supports.items():
        record["all_effective_signatures"] = [
            f"{signature}:{count}" for signature, count in signature_sets[support].most_common()
        ]
        record["selected_family_signatures"] = [
            f"{signature}:{count}" for signature, count in selected_signature_sets[support].most_common()
        ]
        if record["has_selected_family_signature"]:
            record["cover_status"] = "selector_bearing_support_requires_certificate_or_existing_closure"
            record["analytic_endpoint_gap_if_in_strip"] = ""
        else:
            record["cover_status"] = "closed_no_selector_support_exclusion"
            record["analytic_endpoint_gap_if_in_strip"] = endpoint_gap_if_in_strip
        ledger.append(record)

    return sorted(
        ledger,
        key=lambda row: (
            row["has_selected_family_signature"],
            -row["effective_ordered_triad_count"],
            row["support"],
        ),
    )

def leakage_residual_shells(support: tuple[int, ...]) -> set[int]:
    support_set = set(support)
    residual = (support_set & W34_I3_RESIDUAL_SHELLS) | (support_set & W34_I4_RESIDUAL_SHELLS)
    return residual


def support_tuple_from_label(label: str) -> tuple[int, ...]:
    return tuple(int(part) for part in label.split())


def residual_leakage_budget_row(row: dict[str, Any]) -> dict[str, Any]:
    support = support_tuple_from_label(row["support"])
    residual_shells = leakage_residual_shells(support)
    problem = _build_exact_shell_problem(list(support))
    if problem is None:
        raise RuntimeError(f"empty exact shell problem for support {support}")
    shells = np.asarray(problem["k2s"], dtype=np.int64)
    min_shell = float(np.min(shells))
    n_modes = int(problem["N"])
    buckets = {
        slots: {
            "residual_slots": slots,
            "triad_count": 0,
            "coarse_cauchy_coefficient_sum": 0.0,
        }
        for slots in range(4)
    }
    for ell_raw, r_raw, s_raw in zip(
        problem["ell_idx"],
        problem["r_idx"],
        problem["s_idx"],
        strict=True,
    ):
        bases = [int(ell_raw % n_modes), int(r_raw % n_modes), int(s_raw % n_modes)]
        residual_slots = sum(1 for base in bases if int(shells[base]) in residual_shells)
        ell_base, r_base, s_base = bases
        coeff = (
            float(shells[ell_base])
            * math.sqrt(float(shells[s_base]))
            / math.sqrt(
                8.0
                * float(shells[ell_base])
                * float(shells[r_base])
                * float(shells[s_base])
                * min_shell
            )
        )
        buckets[residual_slots]["triad_count"] += 1
        buckets[residual_slots]["coarse_cauchy_coefficient_sum"] += abs(float(coeff))
    c1 = float(buckets[1]["coarse_cauchy_coefficient_sum"])
    c2 = float(buckets[2]["coarse_cauchy_coefficient_sum"])
    c3 = float(buckets[3]["coarse_cauchy_coefficient_sum"])
    return {
        "support": row["support"],
        "residual_shells": " ".join(str(value) for value in sorted(residual_shells)),
        "mode_count": n_modes,
        "triad_count": int(len(problem["ell_idx"])),
        "selected_family_signatures": row["selected_family_signatures"],
        "effective_ordered_triad_count": row["effective_ordered_triad_count"],
        "selected_family_effective_triad_count": row["selected_family_effective_triad_count"],
        "coarse_cauchy_c1_one_residual_slot": c1,
        "coarse_cauchy_c2_two_residual_slots": c2,
        "coarse_cauchy_c3_three_residual_slots": c3,
        "coarse_cauchy_at_delta_0p01": c1 * 0.01**0.5 + c2 * 0.01 + c3 * 0.01**1.5,
        "coarse_cauchy_at_delta_0p05": c1 * 0.05**0.5 + c2 * 0.05 + c3 * 0.05**1.5,
        "coarse_cauchy_at_delta_0p10": c1 * 0.10**0.5 + c2 * 0.10 + c3 * 0.10**1.5,
        "by_residual_slots": [buckets[index] for index in range(4)],
    }


def build_residual_leakage_budget_ledger(
    residual_support_ledger: list[dict[str, Any]],
) -> list[dict[str, Any]]:
    rows = [
        residual_leakage_budget_row(row)
        for row in residual_support_ledger
        if row["has_selected_family_signature"]
    ]
    return sorted(
        rows,
        key=lambda row: (
            -row["coarse_cauchy_c1_one_residual_slot"],
            row["support"],
        ),
    )


def parse_raw_below_logs(pattern: str) -> list[dict[str, Any]]:
    endpoint = Decimal(str(K4_ENDPOINT_THRESHOLD))
    parsed: list[dict[str, Any]] = []
    for path in sorted(Path().glob(pattern)):
        text = path.read_text(errors="replace", encoding="utf-8")
        interval = re.search(
            r"CERTIFIED INTERVAL: C\(shells ([^)]+)\) in \[([^,]+), ([^\]]+)\]",
            text,
        )
        if not interval:
            continue
        support = tuple(int(part.strip()) for part in interval.group(1).split(","))
        lower = Decimal(interval.group(2).strip())
        upper = Decimal(interval.group(3).strip())
        gap = endpoint - upper
        mpmath_value = re.search(r"mpmath R\(x\*\) = ([0-9.eE+\-]+)", text)
        statuses = re.findall(r"STATUS: (.+)", text)
        global_certs = re.findall(r"GLOBAL CERT: (.+)", text)
        projected_gradients = re.findall(
            r"Projected grad(?:ient)? norm\s*:?\s*([0-9.eE+\-]+)",
            text,
        )
        mode_triad = re.search(r"C\(shells [^)]+\) modes N=(\d+), triads T=(\d+)", text)
        active_modes = re.findall(r"Active modes:\s+(\d+)/(\d+)", text)
        hessian_skipped = "Hessian skipped (--no-hessian)" in text
        global_pass = bool(global_certs and global_certs[-1].startswith("PASS"))
        below_endpoint = gap > 0
        if below_endpoint and global_pass and hessian_skipped:
            packaging_status = "raw_below_global_replay_needs_local_hessian_or_interval_packaging"
        elif below_endpoint and global_pass:
            packaging_status = "raw_below_global_replay_with_local_packaging_check_log"
        elif below_endpoint:
            packaging_status = "raw_below_without_global_replay"
        else:
            packaging_status = "not_below_endpoint"
        parsed.append(
            {
                "support": ",".join(str(value) for value in support),
                "log_path": str(path),
                "mode_count": "" if not mode_triad else int(mode_triad.group(1)),
                "triad_count": "" if not mode_triad else int(mode_triad.group(2)),
                "active_modes": "" if not active_modes else f"{active_modes[-1][0]}/{active_modes[-1][1]}",
                "projected_grad_norm": "" if not projected_gradients else projected_gradients[-1],
                "mpmath_R": "" if not mpmath_value else mpmath_value.group(1),
                "certified_lower": str(lower),
                "certified_upper": str(upper),
                "interval_width": str(upper - lower),
                "endpoint_gap_using_certified_upper": str(gap),
                "below_endpoint_by_certified_upper": below_endpoint,
                "hessian_skipped": hessian_skipped,
                "status": "" if not statuses else statuses[-1].strip(),
                "global_cert": "" if not global_certs else global_certs[-1].strip(),
                "packaging_status": packaging_status,
            }
        )
    return sorted(parsed, key=lambda row: Decimal(str(row["endpoint_gap_using_certified_upper"])))


def raw_below_summary(rows: list[dict[str, Any]]) -> dict[str, Any]:
    gaps = [Decimal(str(row["endpoint_gap_using_certified_upper"])) for row in rows]
    return {
        "raw_below_log_count": len(rows),
        "all_below_endpoint": bool(rows) and all(bool(row["below_endpoint_by_certified_upper"]) for row in rows),
        "all_global_replay_pass": bool(rows) and all(str(row["global_cert"]).startswith("PASS") for row in rows),
        "all_hessian_skipped": bool(rows) and all(bool(row["hessian_skipped"]) for row in rows),
        "minimum_endpoint_gap_using_certified_upper": "" if not gaps else str(min(gaps)),
        "minimum_gap_support": "" if not rows else rows[0]["support"],
        "packaging_status_counts": dict(Counter(str(row["packaging_status"]) for row in rows)),
    }


def parse_raw_below_boundary_summaries(
    pattern: str,
    allowed_supports: set[str] | None = None,
) -> list[dict[str, Any]]:
    by_face: dict[tuple[str, str], dict[str, Any]] = {}
    for path in sorted(Path().glob(pattern)):
        payload = read_json(path)
        for row in payload.get("rows", []):
            support = str(row["support"]).replace(",", " ")
            if allowed_supports is not None and support not in allowed_supports:
                continue
            boundary = str(row["boundary"])
            key = (support, boundary)
            record = {
                "support": support,
                "boundary": boundary,
                "R": float(row["R"]),
                "G4": float(row["G4"]),
                "target_G4": float(row["target_G4"]),
                "constraint_abs": float(row["constraint_abs"]),
                "gap_to_endpoint": float(row["gap_to_endpoint"]),
                "interval_endpoint_gap_lower_positive": bool(
                    row.get("interval_value_replay", {}).get("endpoint_gap_lower_positive", False)
                ),
                "summary_json": str(path),
            }
            previous = by_face.get(key)
            if previous is None or record["R"] > float(previous["R"]):
                by_face[key] = record
    return sorted(by_face.values(), key=lambda row: (row["support"], row["boundary"]))


def raw_below_boundary_summary(rows: list[dict[str, Any]]) -> dict[str, Any]:
    expected_face_count = 10
    gaps = [float(row["gap_to_endpoint"]) for row in rows]
    constraints = [float(row["constraint_abs"]) for row in rows]
    tight = min(rows, key=lambda row: float(row["gap_to_endpoint"])) if rows else None
    return {
        "raw_below_boundary_face_count": len(rows),

def merge_residual_tube_certificate_payloads(payloads: list[dict[str, Any]]) -> dict[str, Any]:
    if not payloads:
        return {}
    by_support: dict[str, dict[str, Any]] = {}
    for payload in payloads:
        for row in payload.get("rows", []):
            support = str(row.get("support", "")).replace(",", " ")
            if not support:
                continue
            record = dict(row)
            record["source_summary_json"] = payload.get("summary_json", "")
            violation = float(record.get("violation", 0.0))
            score = float(record.get("R", 0.0)) - 1.0e3 * max(0.0, violation - 1.0e-8)
            prior = by_support.get(support)
            prior_violation = float(prior.get("violation", 0.0)) if prior else 0.0
            prior_score = (
                float(prior.get("R", 0.0)) - 1.0e3 * max(0.0, prior_violation - 1.0e-8)
                if prior
                else -float("inf")
            )
            if prior is None or score > prior_score:
                by_support[support] = record
    rows = [by_support[key] for key in sorted(by_support)]
    return {
        "status": "merged constrained residual-tube certificate summaries",
        "summary_json": "; ".join(payload.get("summary_json", "") for payload in payloads),
        "summary_jsons": [payload.get("summary_json", "") for payload in payloads],
        "endpoint_threshold": payloads[-1].get("endpoint_threshold", ""),
        "delta_max": payloads[-1].get("delta_max", ""),
        "rows": rows,
        "max_R": max((float(row.get("R", 0.0)) for row in rows), default=""),
        "min_gap_to_endpoint": min(
            (float(row.get("gap_to_endpoint", 0.0)) for row in rows),
            default="",
        ),
        "max_constraint_violation": max(
            (float(row.get("violation", 0.0)) for row in rows),
            default="",
        ),
        "all_feasible_with_1e_minus_8": bool(rows)
        and all(float(row.get("violation", 0.0)) <= 1.0e-8 for row in rows),
        "all_feasible_with_1e_minus_7": bool(rows)
        and all(float(row.get("violation", 0.0)) <= 1.0e-7 for row in rows),
    }


def residual_boundary_tangent_summary(payloads: list[dict[str, Any]]) -> dict[str, Any]:
    if not payloads:
        return {}
    by_support: dict[str, dict[str, Any]] = {}
    for payload in payloads:
        for row in payload.get("rows", []):
            if row.get("cert_sign_preserving_l2_radius") in ("", None):
                continue
            support = str(row.get("support", "")).replace(",", " ")
            prior = by_support.get(support)
            record = dict(row)
            record["summary_json"] = payload.get("summary_json", "")
            if prior is None or float(record.get("R", 0.0)) > float(prior.get("R", 0.0)):
                by_support[support] = record
    rows = [by_support[key] for key in sorted(by_support)]
    radii = [float(row["cert_sign_preserving_l2_radius"]) for row in rows]
    weakest = min(rows, key=lambda row: float(row["cert_sign_preserving_l2_radius"])) if rows else None
    return {
        "summary_jsons": [payload.get("summary_json", "") for payload in payloads],
        "certified_row_count": len(rows),
        "certified_supports": [str(row.get("support", "")) for row in rows],
        "all_certified_escape_signs": bool(rows)
        and all(
            float(row.get("cert_sign_preserving_l2_radius", 0.0)) > 0.0
            and abs(float(row.get("full_DG4_F", 0.0))) > 0.0
            for row in rows
        ),
        "minimum_certified_radius": "" if not radii else min(radii),
        "weakest_chart": "" if weakest is None else str(weakest.get("chart", weakest.get("support", ""))),
        "rows": rows,
    }


def candidate_residual_x_fraction(support: tuple[int, ...], npz_path_text: str) -> float | None:
    if not npz_path_text:
        return None
    npz_path = Path(npz_path_text)
    if not npz_path.is_absolute():
        npz_path = REPO_ROOT / npz_path
    if not npz_path.exists():
        return None
    residual_shells = leakage_residual_shells(support)
    if not residual_shells:
        return 0.0
    problem = _build_exact_shell_problem(list(support))
    if problem is None:
        return None
    shells = np.asarray(problem["k2s"], dtype=np.float64)
    try:
        with np.load(npz_path, allow_pickle=True) as data:
            params = np.asarray(data["best_x"], dtype=np.float64)
    except Exception:
        return None
    if params.size != 4 * shells.size:
        return None
    amplitudes = np.exp(params[3::4])
    x_mass = shells * amplitudes
    total = float(np.sum(x_mass))
    if total <= 0.0:
        return None
    mask = np.isin(shells.astype(np.int64), np.asarray(sorted(residual_shells), dtype=np.int64))
    return float(np.sum(x_mass[mask]) / total)


def enrich_single_band_residual_summary(payload: dict[str, Any]) -> dict[str, Any]:
    if not payload:
        return payload
    above_rows: list[dict[str, Any]] = []
    for row in payload.get("support_summaries", []):
        support = tuple(int(value) for value in row.get("support", []))
        row["residual_shells"] = sorted(leakage_residual_shells(support))
        for candidate_key in ("best_feasible", "best_any"):
            candidate = row.get(candidate_key) or {}
            fraction = candidate_residual_x_fraction(support, str(candidate.get("npz", "")))
            if fraction is not None:
                candidate["residual_x_fraction"] = fraction
        gap = row.get("best_feasible_gap_to_endpoint_threshold")
        fraction = (row.get("best_feasible") or {}).get("residual_x_fraction")
        if gap is not None and fraction is not None and float(gap) < 0.0:
            above_rows.append(row)
    if above_rows:
        tight = min(
            above_rows,
            key=lambda row: float((row.get("best_feasible") or {})["residual_x_fraction"]),
        )
        payload["above_endpoint_min_residual_x_fraction"] = float(
            (tight.get("best_feasible") or {})["residual_x_fraction"]
        )
        payload["above_endpoint_min_residual_x_fraction_support"] = " ".join(
            str(value) for value in tight.get("support", [])
        )
    else:
        payload["above_endpoint_min_residual_x_fraction"] = ""
        payload["above_endpoint_min_residual_x_fraction_support"] = ""
    return payload

def write_csv(path: Path, rows: list[dict[str, Any]]) -> None:
    if not rows:
        path.write_text("", encoding="utf-8")
        return
    fields = list(rows[0])
    with path.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        for row in rows:
            writer.writerow(row)

def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--label", default="k34_transition_cover_audit")
    parser.add_argument("--catalogue-dir", type=Path, default=DEFAULT_CATALOGUE_DIR)
    parser.add_argument("--selector-screen-json", type=Path, default=DEFAULT_SELECTOR_SCREEN)
    parser.add_argument("--v13-summary-json", type=Path, default=DEFAULT_V13)
    parser.add_argument("--raw-below-log-glob", default=DEFAULT_RAW_BELOW_LOG_GLOB)
    parser.add_argument("--raw-below-boundary-glob", default=DEFAULT_RAW_BELOW_BOUNDARY_GLOB)
    parser.add_argument("--raw-below-interior-glob", default=DEFAULT_RAW_BELOW_INTERIOR_GLOB)
    parser.add_argument("--raw-below-exterior-glob", default=DEFAULT_RAW_BELOW_EXTERIOR_GLOB)
    parser.add_argument("--skeleton-screen-glob", default=DEFAULT_SKELETON_SCREEN_GLOB)

    parser.add_argument("--single-band-screen-glob", default=DEFAULT_SINGLE_BAND_SCREEN_GLOB)

    parser.add_argument("--residual-boundary-tangent-glob", default=DEFAULT_RESIDUAL_BOUNDARY_TANGENT_GLOB)
    return parser.parse_args()
