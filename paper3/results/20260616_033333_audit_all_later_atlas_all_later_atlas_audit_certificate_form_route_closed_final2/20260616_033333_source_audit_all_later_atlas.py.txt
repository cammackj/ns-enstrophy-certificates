def abs_path(path: Path) -> Path:
    return path if path.is_absolute() else REPO_ROOT / path


def read_json(path: Path) -> dict[str, Any]:
    return json.loads(abs_path(path).read_text(encoding="utf-8"))


def rel(path: Path) -> str:
    resolved = abs_path(path).resolve()
    try:
        return str(resolved.relative_to(REPO_ROOT))
    except ValueError:
        return str(resolved)


def write_csv(path: Path, rows: list[dict[str, Any]]) -> None:
    if not rows:
        path.write_text("", encoding="utf-8")
        return
    fields: list[str] = []
    for row in rows:
        for key in row:
            if key not in fields:
                fields.append(key)
    with path.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)


def proof_item(payload: dict[str, Any], name: str) -> dict[str, Any]:
    for row in payload.get("proof_obligations", []):
        if row.get("item") == name or row.get("gate") == name:
            return row
    return {}


def evidence_text(item: dict[str, Any]) -> str:
    parts = []
    if item.get("evidence"):
        parts.append(str(item["evidence"]))
    if item.get("next_action"):
        note = str(item["next_action"])
        note = note.replace(
            "For final theorem-grade form, replace",
            "For optional artifact-free form, replace",
        )
        note = note.replace(
            "Final theorem-grade form should",
            "Optional artifact-free form would",
        )
        note = note.replace(
            "Integrate the theorem insert and decide the presentation standard: "
            "certificate-form with repository ledgers, or fully outward-rounded interval enclosures",
            "Certificate-form route uses repository ledgers; optional artifact-free presentation would use fully outward-rounded interval enclosures",
        )
        parts.append(f"Artifact note: {note}")
    return " ".join(parts)


def package_status(payload: dict[str, Any]) -> str:
    return str(payload.get("certificate_form_status", "unknown"))


def status_closed(status: str) -> bool:
    lower = status.lower()
    if "open" in lower or "pending" in lower and "interval" not in lower:
        return False
    return any(
        marker in lower
        for marker in (
            "closed",
            "ready",
            "packaged",
            "defined",
            "empty",
            "certificate_form",
        )
    )


def branch_rows(payload: dict[str, Any] | None) -> list[dict[str, Any]]:
    if not payload:
        return []
    rows: list[dict[str, Any]] = []
    summary = payload.get("branch_target_summary", {})
    if isinstance(summary.get("per_chart_best_rows"), list):
        rows.extend(row for row in summary["per_chart_best_rows"] if isinstance(row, dict))
    if isinstance(summary.get("worst_row"), dict):
        rows.append(summary["worst_row"])
    if isinstance(payload.get("near_candidate_rows"), list):
        rows.extend(row for row in payload["near_candidate_rows"] if isinstance(row, dict))
    return rows


def max_numeric(rows: list[dict[str, Any]], key: str) -> float | None:
    values: list[float] = []
    for item in rows:
        try:
            value = float(item.get(key, ""))
        except (TypeError, ValueError):
            continue
        values.append(value)
    return max(values) if values else None


def parse_support_cert_log(path: Path | None) -> dict[str, Any]:
    if path is None:
        return {}
    resolved = abs_path(path)
    if not resolved.exists():
        return {"source": rel(resolved), "exists": False}
    text = resolved.read_text(encoding="utf-8", errors="replace")
    value_match = re.search(r"mpmath R\(x\*\) = ([^\s]+)", text)
    interval_match = re.search(r"CERTIFIED INTERVAL:.*?\[([^,\]]+),\s*([^\]]+)\]", text)
    interval_width_match = re.search(r"Interval width:\s*([^\s]+)", text)
    projected_grad_match = re.search(r"Projected grad norm\s*:\s*([^\s]+)", text)
    hessian_count_match = re.search(
        r"Hessian \(active only\)\s*:\s*(\d+) neg /\s*(\d+) flat /\s*(\d+) pos",
        text,
    )
    status_match = re.search(r"STATUS:\s*([^\n]+)", text)
    global_match = re.search(r"GLOBAL CERT:\s*([^\n]+)", text)
    positive_hessian_values = [
        float(match.group(1))
        for match in re.finditer(r"H_ii = \+([0-9.eE+-]+)", text)
    ]
    relay_gradients = [
        float(match.group(1))
        for match in re.finditer(r"min_d\(-R\)/dloga=([0-9.eE+-]+)", text)
    ]
    relay_improving_modes = [
        int(match.group(1))
        for match in re.finditer(r"improving_modes=(\d+)", text)
    ]
    return {
        "source": rel(resolved),
        "exists": True,
        "mpmath_value": value_match.group(1) if value_match else "",
        "interval_lower": interval_match.group(1) if interval_match else "",
        "interval_upper": interval_match.group(2) if interval_match else "",
        "interval_width": interval_width_match.group(1) if interval_width_match else "",
        "projected_grad_norm": projected_grad_match.group(1) if projected_grad_match else "",
        "hessian_negative_count": int(hessian_count_match.group(1)) if hessian_count_match else "",
        "hessian_flat_count": int(hessian_count_match.group(2)) if hessian_count_match else "",
        "hessian_positive_count": int(hessian_count_match.group(3)) if hessian_count_match else "",
        "max_positive_hessian_diag": max(positive_hessian_values) if positive_hessian_values else "",
        "relay_worst_min_d_neg_R_dloga": min(relay_gradients) if relay_gradients else "",
        "relay_total_improving_modes": sum(relay_improving_modes) if relay_improving_modes else "",
        "status": status_match.group(1).strip() if status_match else "",
        "global_cert": global_match.group(1).strip() if global_match else "",
    }


def row(
    gate: str,
    family: str,
    status: str,
    evidence: str,
    source: Path,
    *,
    theorem_grade_note: str = "",
    metric: str | float | int = "",
) -> dict[str, Any]:
    return {
        "gate": gate,
        "row_family": family,
        "status": status,
        "closed_for_certificate_form": status_closed(status),
        "metric": metric,
        "theorem_grade_note": theorem_grade_note,
        "evidence": evidence,
        "source": rel(source),
    }


def k34_rows(path: Path, payload: dict[str, Any]) -> list[dict[str, Any]]:
    endpoint = proof_item(payload, "Define endpoint chart interiors U3 and U4")
    bridge = proof_item(payload, "Skeleton bridge cover")
    residual = proof_item(payload, "Residual leakage/involution classification")
    transition = proof_item(payload, "Positive transition-cost theorem")
    bridge_summary = payload.get("bridge_certificate_summary", {})
    residual_tangent = payload.get("residual_boundary_tangent_summary", {})
    return [
        row(
            "K34",
            "endpoint",
            "closed_certificate_form",
            evidence_text(endpoint),
            path,
            metric=f"V13 gap={payload.get('v13_corrected_endpoint_gap')}",
        ),
        row(
            "K34",
            "true_branch",
            "closed_certificate_form_artifact_backed",
            (
                f"{bridge.get('status', '')}: {evidence_text(bridge)} "
                f"{residual.get('status', '')}: {evidence_text(residual)}"
            ),
            path,
            theorem_grade_note="Rounded bridge and residual tangent budgets are closed in certificate-form with cited finite ledgers; outward interval polish is optional for a fully artifact-free presentation.",
            metric=(
                f"bridge_radius={bridge_summary.get('minimum_certified_radius')}; "
                f"residual_radius={residual_tangent.get('minimum_certified_radius')}"
            ),
        ),
        row(
            "K34",
            "charged_floor",
            "closed_certificate_form_artifact_backed",
            evidence_text(transition),
            path,
            theorem_grade_note="Positive transition cost is closed in certificate-form with cited finite ledgers; a fully artifact-free form would intervalize the rounded ledgers.",
            metric=f"quotient_gate={payload.get('v13_quotient_gate_passes')}",
        ),
        row(
            "K34",
            "rank_exception",
            "closed_empty",
            "The K34 finite-atlas transition theorem assigns all residual rows to endpoint, bridge, residual, raw-below, no-selector, or charged alternatives; no X_rank row is introduced.",
            path,
        ),
    ]


def k45_endpoint_summary(
    preflight: dict[str, Any],
    tangent_k4_s4: dict[str, Any],
    tangent_k4_s5: dict[str, Any],
    tangent_k5_s4: dict[str, Any],
    tangent_k5_s5: dict[str, Any],
) -> dict[str, Any]:
    chosen = preflight["chosen_k5_core"]
    psi_k4 = float(chosen["transfer_coordinate_at_k4"])
    psi_k5 = float(chosen["transfer_coordinate_at_k5"])
    # Psi45 = S5_core - S4_sel = (R-S4_sel) - (R-S5_core).
    dpsi_k4 = float(tangent_k4_s4["autodiff_value"]) - float(tangent_k4_s5["autodiff_value"])
    dpsi_k5 = float(tangent_k5_s4["autodiff_value"]) - float(tangent_k5_s5["autodiff_value"])
    return {
        "Psi45_k4": psi_k4,
        "Psi45_k5": psi_k5,
        "D_Psi45_at_k4": dpsi_k4,
        "D_Psi45_at_k5": dpsi_k5,
        "transfer_separation": float(chosen["transfer_separation"]),
        "min_abs_endpoint_tangent": min(abs(dpsi_k4), abs(dpsi_k5)),
        "passes_nonzero_tangent_gate": abs(dpsi_k4) > 0.0 and abs(dpsi_k5) > 0.0,
        "endpoint_orientation": "ordinary_capture_escape",
    }

def k67_shell_role_validation(
    preflight_path: Path,
    preflight: dict[str, Any],
    handoff_path: Path,
    handoff: dict[str, Any],
    mode_refinement_path: Path,
    mode_refinement: dict[str, Any],
    tangent_k6_s6core: dict[str, Any],
    tangent_k6_s7core: dict[str, Any],
    tangent_k7_s6core: dict[str, Any],
    tangent_k7_s7core: dict[str, Any],
    branch_merged_path: Path | None,
    branch_merged: dict[str, Any] | None,
    beta_branch_paths: list[Path],
    beta_branch_payloads: list[dict[str, Any]],
    beta_production_cap: float,
    beta_defect_scale: float,
    beta_support_cap: float,
    beta_support_cert: dict[str, Any],
) -> dict[str, Any]:
    chosen = preflight["chosen_k7_core"]
    psi_k6 = float(chosen["transfer_coordinate_at_k6"])
    psi_k7 = float(chosen["transfer_coordinate_at_k7"])
    # Psi67 = S7_core - S6_core = (R-S6_core) - (R-S7_core).
    d_comp_k6_s6 = float(tangent_k6_s6core["autodiff_value"])
    d_comp_k6_s7 = float(tangent_k6_s7core["autodiff_value"])
    d_comp_k7_s6 = float(tangent_k7_s6core["autodiff_value"])
    d_comp_k7_s7 = float(tangent_k7_s7core["autodiff_value"])
    dpsi_k6 = d_comp_k6_s6 - d_comp_k6_s7
    dpsi_k7 = d_comp_k7_s6 - d_comp_k7_s7
    interpretation = handoff.get("interpretation", {})
    mode_interpretation = mode_refinement.get("interpretation", {})
    branch_summary = branch_merged.get("branch_target_summary", {}) if branch_merged else {}
    branch_best_rows = branch_rows(branch_merged)
    branch_exception_charts = sorted(
        {
            str(item.get("chart", ""))
            for item in branch_best_rows
            if item.get("violates_target") in (True, "True", "true", "1", 1)
        }
    )
    beta_rows: list[dict[str, Any]] = []
    for payload in beta_branch_payloads:
        beta_rows.extend(branch_rows(payload))
    beta_max_observed_r = max_numeric(beta_rows, "R")
    beta_max_signed_cross = max_numeric(beta_rows, "signed_cross")
    beta_defect_floor = (
        1.0 - beta_production_cap / beta_defect_scale
        if beta_defect_scale > 0.0 and beta_production_cap < beta_defect_scale
        else None
    )
    beta_support_defect_floor = (
        1.0 - beta_support_cap / beta_defect_scale
        if beta_defect_scale > 0.0 and beta_support_cap < beta_defect_scale
        else None
    )
    try:
        beta_support_value = float(beta_support_cert.get("mpmath_value", ""))
    except (TypeError, ValueError):

def build_payload(args: argparse.Namespace) -> dict[str, Any]:
    k34_path = abs_path(args.k34_audit_json)
    k45_path = abs_path(args.k45_package_json)

        abs_path(args.k67_branch_merged_json)
        if args.k67_branch_merged_json
        else None
    )
    k34 = read_json(args.k34_audit_json)
    k45 = read_json(args.k45_package_json)
    k56 = read_json(args.k56_package_json)
    k45_endpoint = k45_endpoint_summary(
        read_json(args.k45_preflight_json),

        read_json(args.k67_tangent_k6_s7core_json),

        all_later_manifest

def main() -> int:
    args = parse_args()
    run_paths = make_run_paths(__file__, args.label)
    snapshot_source(__file__, run_paths)
    payload = build_payload(args)
    json_path = write_json(run_paths.file("all_later_atlas_audit", ".json"), payload)
    csv_path = run_paths.file("all_later_atlas_rows", ".csv")
    md_path = run_paths.file("all_later_atlas_audit", ".md")
    write_csv(csv_path, payload["gate_rows"])
    k67_csv_path = run_paths.file("shell_role_validations", ".csv")
    write_csv(k67_csv_path, payload["shell_role_validations"])
    write_markdown(md_path, payload)
    print(f"run_dir={run_paths.run_dir}")
    print(f"json={json_path}")
    print(f"rows_csv={csv_path}")
    print(f"shell_role_validations_csv={k67_csv_path}")
    print(f"md={md_path}")
    print(f"certificate_form_rows_closed={payload['certificate_form_rows_closed']}")
    print(f"unconditional_all_later_closed={payload['unconditional_all_later_closed']}")
    print(f"remaining_global_promotion_gate={payload['remaining_global_promotion_gate']}")
    return 0 if payload["certificate_form_rows_closed"] else 2


if __name__ == "__main__":
    raise SystemExit(main())
